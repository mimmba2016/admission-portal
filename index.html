Google Apps Script (`Code.gs`)

Paste this script in Google Sheet's Extensions > Apps Script to enable 24/7 synchronization and save all 5 attachments into Google Drive. 

function doPost(e) {
  try {
    var raw = JSON.parse(e.postData.contents);
    var action = raw.action;
    var state = raw.state;
    if (action === "syncFullState" && state) {
      syncStateToGoogleSheets(state);
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
